## Decisions this plan makes where the design is silent

1. **An empty `"metadata":{}` is refused** with `invalid_field` / `metadata`. `omitempty` would drop it on re-encode, so accepting it would let a value change bytes while crossing Core. This is the same rule as `validateBlocks` refusing `[]`. A Go-constructed empty map is harmless: it encodes as absent and `Validate()` accepts it.
2. **`*MetadataValidationError` unwraps to `*RequestValidationError{invalid_field, "metadata"}`.** Host's `createbody.classify` (`host/internal/createbody/createbody.go:124-140`) treats any error that is not a `RequestValidationError` as *unsupported*, which means **block**. A metadata limit failure is a malformed body that no Host can read, so it must classify as **reject**. The unwrap gives that result with no change to `RequestValidationError`.
3. **`MetadataValidationError.Key` is set only when the key satisfies the key grammar**, and `Error()` never includes the key or the value. This follows the package rule that caller-chosen bytes are never echoed (`errors.go:12-15`, `:276-280`).
4. **Principal errors name `principal.<member>`**, following the `"error.code"` precedent at `errors.go:82`. A nested unknown member is reported as `unknown_field` / `principal`, and a non-object as `invalid_field` / `principal`.
5. **`metadata` on interrupt, restore or gate_response is `unknown_field` with an empty Field.** It is not a declared member there, and the Go types have no `Metadata` field.
6. **Principal is a pointer.** `InterruptRequest` and `RestoreRequest` are comparable today, and a pointer keeps them comparable. A map field would not, and apidiff would flag that as a break.
7. **`hostlink.attribution.principal` is a `hostlink.<area>.<feature>` token** (area `attribution`), so it does not collide with the `hostlink.command.<kind>` namespace reserved for runtime command kinds (`hostlink_framing.go:72-79`). The doc comment's second bullet names it as an example; no command-kind reservation is needed. (Renamed from `hostlink.command.principal` on 2026-09-25.)
8. **There is no CHANGELOG file.** `core` has never had one (verified: there is no `CHANGELOG*` at the repo root). Release notes go in the annotated tag, as in v0.11.0. Do not create a CHANGELOG.

```go
package v1

import (
	"encoding/json"
	"errors"
)

// PrincipalKind classifies the verified sender of a command.
type PrincipalKind string

const (
	// PrincipalKindActor is a human user acting through a client.
	PrincipalKindActor PrincipalKind = "actor"
	// PrincipalKindService is a non-human service account.
	PrincipalKindService PrincipalKind = "service"
)

// principalMembers is the single declared member list for Principal.
var principalMembers = []string{"tenant", "subject", "kind"}

// Principal is the VERIFIED sender of a command as a Factory established it.
//
// It is never client-supplied: a Factory stamps it after credential
// verification and refuses any request that already carries one. It holds only
// verified facts; a display name is resolved from Subject by the application and
// is never part of identity. Core validates its shape only and proves nothing
// about who the principal is.
type Principal struct {
	Tenant  TenantID      `json:"tenant"`
	Subject SubjectID     `json:"subject"`
	Kind    PrincipalKind `json:"kind"`
}

// Validate reports whether every member is a valid identity and Kind is known.
// A failure is a *RequestValidationError naming principal.tenant,
// principal.subject or principal.kind.
func (p Principal) Validate() error {
	if err := p.Tenant.Validate(); err != nil {
		return invalidRequest(RequestValidationCodeInvalidField, "principal.tenant")
	}
	if err := p.Subject.Validate(); err != nil {
		return invalidRequest(RequestValidationCodeInvalidField, "principal.subject")
	}
	switch p.Kind {
	case PrincipalKindActor, PrincipalKindService:
		return nil
	default:
		return invalidRequest(RequestValidationCodeInvalidField, "principal.kind")
	}
}

// MarshalJSON validates p and encodes its members in the order tenant, subject,
// kind, so an invalid principal can never be written.
func (p Principal) MarshalJSON() ([]byte, error) {
	if err := p.Validate(); err != nil {
		return nil, err
	}
	type principalWire struct {
		Tenant  TenantID      `json:"tenant"`
		Subject SubjectID     `json:"subject"`
		Kind    PrincipalKind `json:"kind"`
	}
	return json.Marshal(principalWire(p))
}

// UnmarshalJSON strictly decodes a principal: an unknown, duplicate, missing or
// null member, or an unknown kind, is refused. Errors name principal.<member>,
// or principal when no member can be named.
func (p *Principal) UnmarshalJSON(data []byte) error {
	fields, err := decodeRequestFields(data, principalMembers...)
	if err != nil {
		return principalFieldError(err)
	}
	tenant, err := decodeRequiredString(fields, "tenant")
	if err != nil {
		return principalFieldError(err)
	}
	subject, err := decodeRequiredString(fields, "subject")
	if err != nil {
		return principalFieldError(err)
	}
	kind, err := decodeRequiredString(fields, "kind")
	if err != nil {
		return principalFieldError(err)
	}
	decoded := Principal{Tenant: TenantID(tenant), Subject: SubjectID(subject), Kind: PrincipalKind(kind)}
	if err := decoded.Validate(); err != nil {
		return err
	}
	*p = decoded
	return nil
}

// principalFieldError re-attributes a member-level failure to the principal's
// member path. A Field reaching here is a contract member name, because
// decodeContractFields redacts every other name, so no caller-chosen bytes are
// echoed. A body that is not an object is invalid_field: the enclosing request
// was already valid JSON.
func principalFieldError(err error) error {
	var validation *RequestValidationError
	if !errors.As(err, &validation) {
		return invalidRequest(RequestValidationCodeInvalidField, "principal")
	}
	code := validation.Code
	if code == RequestValidationCodeInvalidJSON {
		code = RequestValidationCodeInvalidField
	}
	if validation.Field == "" {
		return invalidRequest(code, "principal")
	}
	return invalidRequest(code, "principal."+validation.Field)
}

// decodeOptionalPrincipal decodes a request's optional principal member. Absent
// is nil; an explicit null is invalid_field.
func decodeOptionalPrincipal(fields map[string]json.RawMessage) (*Principal, error) {
	raw, ok := fields["principal"]
	if !ok {
		return nil, nil
	}
	if isJSONNull(raw) {
		return nil, invalidRequest(RequestValidationCodeInvalidField, "principal")
	}
	var principal Principal
	if err := principal.UnmarshalJSON(raw); err != nil {
		return nil, err
	}
	return &principal, nil
}

// validateOptionalPrincipal validates a present principal; absence is valid.
func validateOptionalPrincipal(principal *Principal) error {
	if principal == nil {
		return nil
	}
	return principal.Validate()
}
```

```go
package v1

import (
	"encoding/json"
	"maps"
	"slices"
	"strings"
	"unicode/utf8"
)

// MessageMetadata is the client-supplied, app-defined bag of string fields a
// create or input command may carry. Core bounds and shapes it and assigns no
// key any meaning. It is not identity: the verified sender is Principal, never
// a key here.
type MessageMetadata map[string]string

const (
	// MaxMetadataFields is the largest number of fields one command may carry.
	MaxMetadataFields = 16
	// MaxMetadataKeyBytes is the longest key; keys match ^[a-z][a-z0-9_]{0,63}$.
	MaxMetadataKeyBytes = 64
	// MaxMetadataValueBytes is the longest value, in UTF-8 bytes.
	MaxMetadataValueBytes = 1024
	// MaxMetadataBytes bounds the canonical (encoding/json) encoding of the
	// whole metadata object.
	MaxMetadataBytes = 4096
)

// reservedMetadataKeyPrefix is held for Looprig; a key beginning with it is
// refused. It has no semantics yet.
const reservedMetadataKeyPrefix = "looprig"

// MetadataValidationCode is a stable machine-readable reason metadata is invalid.
type MetadataValidationCode string

const (
	MetadataValidationCodeTooManyFields MetadataValidationCode = "too_many_fields"
	MetadataValidationCodeInvalidKey    MetadataValidationCode = "invalid_key"
	MetadataValidationCodeReservedKey   MetadataValidationCode = "reserved_key"
	MetadataValidationCodeValueTooLarge MetadataValidationCode = "value_too_large"
	MetadataValidationCodeInvalidValue  MetadataValidationCode = "invalid_value"
	MetadataValidationCodeTooLarge      MetadataValidationCode = "too_large"
)

// MetadataValidationError reports invalid MessageMetadata.
//
// Key names the offending key only when the key itself satisfies the key
// grammar (reserved_key, value_too_large, invalid_value); it is empty for
// invalid_key, too_many_fields and too_large, so caller-chosen bytes that fail
// the grammar are never retained. Error never includes Key or any value.
//
// It unwraps to a *RequestValidationError with Code invalid_field and Field
// "metadata", so a consumer that classifies request failures by that type
// treats every metadata failure as a malformed request.
type MetadataValidationError struct {
	Code MetadataValidationCode
	Key  string
}

func (e *MetadataValidationError) Error() string {
	return "sessionwire/v1: invalid request metadata: " + string(e.Code)
}

// Unwrap returns the equivalent request-level validation failure.
func (e *MetadataValidationError) Unwrap() error {
	return invalidRequest(RequestValidationCodeInvalidField, "metadata")
}

// Validate reports whether m is within every limit. Keys are checked in sorted
// order so the reported failure is deterministic. A nil or empty m is valid:
// it encodes as an absent member.
func (m MessageMetadata) Validate() error {
	if len(m) > MaxMetadataFields {
		return &MetadataValidationError{Code: MetadataValidationCodeTooManyFields}
	}
	for _, key := range slices.Sorted(maps.Keys(m)) {
		if !metadataKeyValid(key) {
			return &MetadataValidationError{Code: MetadataValidationCodeInvalidKey}
		}
		if strings.HasPrefix(key, reservedMetadataKeyPrefix) {
			return &MetadataValidationError{Code: MetadataValidationCodeReservedKey, Key: key}
		}
		value := m[key]
		if len(value) > MaxMetadataValueBytes {
			return &MetadataValidationError{Code: MetadataValidationCodeValueTooLarge, Key: key}
		}
		if !metadataValueValid(value) {
			return &MetadataValidationError{Code: MetadataValidationCodeInvalidValue, Key: key}
		}
	}
	encoded, err := json.Marshal(map[string]string(m))
	if err != nil {
		return err
	}
	if len(encoded) > MaxMetadataBytes {
		return &MetadataValidationError{Code: MetadataValidationCodeTooLarge}
	}
	return nil
}

func metadataKeyValid(key string) bool {
	if len(key) == 0 || len(key) > MaxMetadataKeyBytes || key[0] < 'a' || key[0] > 'z' {
		return false
	}
	for index := 1; index < len(key); index++ {
		switch char := key[index]; {
		case char >= 'a' && char <= 'z', char >= '0' && char <= '9', char == '_':
		default:
			return false
		}
	}
	return true
}

func metadataValueValid(value string) bool {
	if !utf8.ValidString(value) {
		return false
	}
	for _, char := range value {
		if char < 0x20 && char != '\t' && char != '\n' {
			return false
		}
	}
	return true
}

// reportableMetadataKey returns key when it satisfies the key grammar, else "".
func reportableMetadataKey(key string) string {
	if metadataKeyValid(key) {
		return key
	}
	return ""
}

// decodeOptionalMetadata decodes a request's optional metadata member. Absent
// is nil. An explicit null, a non-object and an empty object are invalid_field:
// an empty object would re-encode as absent, changing the bytes. Every value
// must be a JSON string. Limits are applied by MessageMetadata.Validate, which
// the enclosing request's Validate calls.
func decodeOptionalMetadata(fields map[string]json.RawMessage) (MessageMetadata, error) {
	raw, ok := fields["metadata"]
	if !ok {
		return nil, nil
	}
	if isJSONNull(raw) {
		return nil, invalidRequest(RequestValidationCodeInvalidField, "metadata")
	}
	members, err := decodeJSONObject(raw)
	if err != nil {
		return nil, invalidNestedJSONObject(err, "metadata")
	}
	if len(members) == 0 {
		return nil, invalidRequest(RequestValidationCodeInvalidField, "metadata")
	}
	metadata := make(MessageMetadata, len(members))
	for _, key := range slices.Sorted(maps.Keys(members)) {
		value, err := decodeStrictJSONString(members[key])
		if err != nil {
			return nil, &MetadataValidationError{Code: MetadataValidationCodeInvalidValue, Key: reportableMetadataKey(key)}
		}
		metadata[key] = value
	}
	return metadata, nil
}
```

```go
// FuzzInputRequestMessageMembers: any body Core accepts validates, and its
// canonical encoding is a fixed point (decode(encode(x)) re-encodes identically).
func FuzzInputRequestMessageMembers(f *testing.F) {
	for _, seed := range []string{
		`{"version":1,"command_id":"c","session_id":"s","blocks":[{"type":"text","text":"x"}]}`,
		`{"version":1,"command_id":"c","session_id":"s","blocks":[{"type":"text","text":"x"}],"metadata":{"space":"family & friends"},"principal":{"tenant":"t","subject":"u","kind":"actor"}}`,
		`{"version":1,"command_id":"c","session_id":"s","blocks":[{"type":"text","text":"x"}],"metadata":{"a":" "}}`,
	} {
		f.Add([]byte(seed))
	}
	f.Fuzz(func(t *testing.T, data []byte) {
		var first sessionwire.InputRequest
		if err := json.Unmarshal(data, &first); err != nil {
			return
		}
		if err := first.Validate(); err != nil {
			t.Fatalf("decoded request fails Validate: %v", err)
		}
		encoded, err := json.Marshal(first)
		if err != nil {
			t.Fatalf("marshal decoded request: %v", err)
		}
		var second sessionwire.InputRequest
		if err := json.Unmarshal(encoded, &second); err != nil {
			t.Fatalf("re-decode canonical encoding %s: %v", encoded, err)
		}
		again, err := json.Marshal(second)
		if err != nil {
			t.Fatalf("re-marshal: %v", err)
		}
		if !bytes.Equal(encoded, again) {
			t.Fatalf("canonical encoding is not a fixed point\nfirst:  %s\nsecond: %s", encoded, again)
		}
	})
}
```
